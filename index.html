async function submitAllAnswers() { const btn = document.getElementById('next-btn'); btn.disabled = true; btn.innerText = 'Enregistrement en cours...'; document.getElementById('error-box').classList.add('hidden'); try { // 1. Détection automatique du nom de la table let nomTableReponses = 'Reponses'; try { await grist.docApi.fetchTable('Reponses'); } catch(e) { nomTableReponses = 'Réponses'; } // 2. Création de l'utilisateur avec ses champs propres const resUser = await grist.docApi.applyUserActions([ ['AddRecord', 'Utilisateurs', null, { Nom: managerInfos.Nom, Prenom: managerInfos.Prenom || "", Prénom: managerInfos.Prenom || "", Fonction: managerInfos.Fonction || "", Email: managerInfos.Email || "", Date_Inscription: managerInfos.Date_Inscription }] ]); let newUserId = null; if (Array.isArray(resUser) && resUser.length > 0) { newUserId = (typeof resUser[0] === 'object' && resUser[0] !== null) ? (resUser[0].id || resUser[0].rowId) : resUser[0]; } else if (resUser && typeof resUser === 'object') { newUserId = resUser.id || resUser.rowId; } else { newUserId = resUser; } if (!newUserId) { const tUsers = await grist.docApi.fetchTable('Utilisateurs'); newUserId = tUsers.id[tUsers.id.length - 1]; } // 3. Envoi des réponses : UNIQUEMENT Utilisateur_ID, Question_ID et Note // Score est calculé automatiquement par Grist via sa propre formule const actions = QUESTIONS.map(q => { return ['AddRecord', nomTableReponses, null, { Utilisateur_ID: newUserId, Question_ID: q.id, Note: userAnswers[q.id] || "1 - Pas d'accord" }]; }); await grist.docApi.applyUserActions(actions); document.getElementById('step-quiz').classList.add('hidden'); document.getElementById('step-success').classList.remove('hidden'); } catch (err) { console.error("Erreur détaillée :", err); showError("Erreur : " + (err.message || JSON.stringify(err))); btn.disabled = false; btn.innerText = 'Réessayer'; } }
